---
title: Quickstart
description: Create a current devstack config and wire the build integrations.
---

Start with a `devstack.config.ts` at the app root:

```ts title="devstack.config.ts"
import { defineDevstack, account, localPackage, sui, wallet } from '@mysten-incubation/devstack';

export const localnet = sui();
export const publisher = account('publisher');
export const alice = account('alice');
export const bob = account('bob');

export const hello = localPackage('hello', {
	sourcePath: './move/hello',
	publisher,
});
export const devWallet = wallet({ accounts: [publisher, alice, bob] });

export default defineDevstack({
	members: [localnet, hello, devWallet],
	stackName: 'quickstart',
	codegen: { outputDir: 'src/generated' },
});
```

Add a Vite config that uses the current build integration:

```ts title="vite.config.ts"
import react from '@vitejs/plugin-react';
import { defineDevstackViteConfig } from '@mysten-incubation/devstack/vite';

export default defineDevstackViteConfig({
	port: 5173,
	plugins: [react()],
});
```

The generated files are importable from application code. For dapp-kit wiring, consume the generated
config file and the generated Sui network file:

```ts title="src/dapp-kit.ts"
import { createDAppKit } from '@mysten/dapp-kit-react';
import { devWalletInitializer } from '@mysten-incubation/dev-wallet';
import { DevstackSignerAdapter, parseDevstackToken } from '@mysten-incubation/dev-wallet/adapters';
import { SuiGrpcClient } from '@mysten/sui/grpc';

import { dappKitConfig } from './generated/dapp-kit/config.js';
import { suiNetwork } from './generated/sui/network.js';

export const dAppKit = createDAppKit({
	networks: [dappKitConfig.chain],
	defaultNetwork: dappKitConfig.chain,
	createClient() {
		return new SuiGrpcClient({
			network: dappKitConfig.chain,
			baseUrl: suiNetwork.rpcUrl,
		});
	},
	walletInitializers: [
		devWalletInitializer({
			adapters: [
				new DevstackSignerAdapter({
					serverOrigin: dappKitConfig.walletUrl,
					token: parseDevstackToken(dappKitConfig.pairUrl),
					name: 'Devstack',
				}),
			],
			autoConnect: true,
			createInitialAccount: false,
		}),
	],
});

(globalThis as { __devstackDAppKit__?: typeof dAppKit }).__devstackDAppKit__ = dAppKit;
```

Run the stack:

```bash
pnpm devstack up
```

For one-shot setup in CI or before typechecking:

```bash
pnpm devstack apply
```

Use the generated files from `src/generated`, and treat `.devstack/` as runtime state only.
